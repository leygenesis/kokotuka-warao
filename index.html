<script>

    // Base de datos de frases completas

    const phrasesDB = {

        "no hablamos mal sobre esa persona ni de su decision": {

            translation: "Oko tai warao asida dibianaja, a obojona asida dibianaja mita.",

            breakdown: [

                { warao: "Oko", es: "Nosotros" },

                { warao: "Tai warao", es: "Esa persona" },

                { warao: "Asida", es: "Mal / Feo" },

                { warao: "Dibianaja", es: "No hablamos" },

                { warao: "A obojona", es: "Su decisión" },

                { warao: "Mita", es: "Ni / Tampoco" }

            ]

        },

        "a todos nos gusta hablar pero podemos hablar de otros con respeto tambien": {

            translation: "Oko kokotuka dibia obonoa, taikaja oko warao rai yakera dibia ja mita.",

            breakdown: [

                { warao: "Oko kokotuka", es: "Todos nosotros" },

                { warao: "Dibia obonoa", es: "Nos gusta hablar" },

                { warao: "Taikaja", es: "Pero / Sin embargo" },

                { warao: "Warao rai", es: "De los demás" },

                { warao: "Yakera", es: "Con respeto / Bien" },

                { warao: "Dibia ja", es: "Podemos hablar" },

                { warao: "Mita", es: "También" }

            ]

        },

        "hola, ¿cómo estás?": {

            translation: "Yakaraja, ¿katukane ja?",

            breakdown: [

                { warao: "Yakaraja", es: "Hola / Bien" },

                { warao: "Katukane ja", es: "¿Cómo estás?" }

            ]

        },

        "gracias": {

            translation: "Yakera",

            breakdown: [

                { warao: "Yakera", es: "Gracias / Está bien" }

            ]

        }

    };

 

    // Diccionario de términos sueltos

    const dictionary = [

        { es: "Nosotros", warao: "Oko" },

        { es: "Todos", warao: "Kokotuka" },

        { es: "Hablar", warao: "Dibia" },

        { es: "Querer / Gustar", warao: "Obonoa" },

        { es: "Bien / Respeto", warao: "Yakera" },

        { es: "Persona", warao: "Warao" },

        { es: "Malo / Feo", warao: "Asida" },

        { es: "Decisión / Pensamiento", warao: "Obojona" },

        { es: "Pero", warao: "Taikaja" },

        { es: "También / Tampoco", warao: "Mita" }

    ];

 

    // Inicializar lista de vocabulario

    function loadDictionary() {

        const dictContainer = document.getElementById('dictList');

        dictContainer.innerHTML = dictionary.map(item => `

            <div class="dict-item">

                <strong>${item.warao}</strong><br>

                <span>${item.es}</span>

            </div>

        `).join('');

    }

 

    function checkEnter(event) {

        if (event.key === "Enter") {

            translateText();

        }

    }

 

    function setPreset(text) {

        document.getElementById('searchInput').value = text;

        translateText();

    }

 

    function translateText() {

        const input = document.getElementById('searchInput').value.trim().toLowerCase();

        const output = document.getElementById('translationOutput');

        const breakdownSection = document.getElementById('breakdownSection');

        const breakdownGrid = document.getElementById('breakdownGrid');

 

        if (!input) {

            output.innerText = "Por favor ingresa un texto.";

            breakdownSection.style.display = "none";

            return;

        }

 

        // Buscar frase completa

        if (phrasesDB[input]) {

            const data = phrasesDB[input];

            output.innerText = data.translation;

           

            breakdownGrid.innerHTML = data.breakdown.map(item => `

                <div class="word-item">

                    <strong>${item.warao}</strong>

                    ${item.es}

                </div>

            `).join('');

           

            breakdownSection.style.display = "block";

            return;

        }

 

        // Buscar término individual en diccionario

        const foundWord = dictionary.find(d => d.es.toLowerCase() === input || d.warao.toLowerCase() === input);

        if (foundWord) {

            output.innerText = `${foundWord.warao} (${foundWord.es})`;

            breakdownSection.style.display = "none";

            return;

        }

 

        // Si no se encuentra resultado exacto

        output.innerText = "Frase no encontrada en la base de datos de ejemplo.";

        breakdownSection.style.display = "none";

    }

 

    // Cargar componentes al iniciar

    loadDictionary();

</script>
